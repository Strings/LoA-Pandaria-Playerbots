cmake_minimum_required(VERSION 3.16)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(UltimateMoPUnitTests LANGUAGES CXX)
  option(BUILD_TESTING "Build server-independent CTest unit tests" ON)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  if(BUILD_TESTING)
    enable_testing()
  endif()
endif()

if(NOT BUILD_TESTING)
  return()
endif()

add_library(ump_test_support INTERFACE)
target_include_directories(ump_test_support INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/framework")

function(ump_add_unit_test name)
  add_executable(${name} ${ARGN})
  target_link_libraries(${name} PRIVATE ump_test_support)
  target_compile_features(${name} PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(${name} PRIVATE /W4)
  else()
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES LABELS "unit")
endfunction()

# Each module owns its test registration; adding a suite never edits the root build.
file(GLOB suites CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "*/CMakeLists.txt")
foreach(suite IN LISTS suites)
  get_filename_component(directory "${suite}" DIRECTORY)
  add_subdirectory("${directory}")
endforeach()
